cmake_minimum_required(VERSION 3.22)
project(fieldguide LANGUAGES C CXX)
foreach(lib llama ggml ggml-base)
    add_library(${lib} SHARED IMPORTED)
    set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${LLAMA_BIN}/lib${lib}.so")
endforeach()
add_library(fieldguide SHARED ask.cpp)
target_include_directories(fieldguide PRIVATE "${LLAMA_DIR}/include" "${LLAMA_DIR}/ggml/include")
target_link_libraries(fieldguide PRIVATE llama ggml ggml-base android log dl)
target_compile_features(fieldguide PRIVATE cxx_std_17)
target_link_options(fieldguide PRIVATE "-Wl,-z,max-page-size=16384")
